cmake_minimum_required(VERSION 3.20)
project(EXaCTz VERSION 0.1.0 LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
option(EXACTZ_BUILD_SINGLE "Build single-GPU research driver" ON)
option(EXACTZ_BUILD_MPI "Build distributed research driver" ON)
find_package(CUDAToolkit REQUIRED)
find_package(OpenMP REQUIRED COMPONENTS CXX)
find_path(SZ3_INCLUDE_DIR SZ3/api/sz.hpp REQUIRED)
find_path(ZSTD_INCLUDE_DIR zstd.h REQUIRED)
find_library(ZSTD_LIBRARY NAMES zstd REQUIRED)
find_path(ZFP_INCLUDE_DIR zfp.h REQUIRED)
find_library(ZFP_LIBRARY NAMES zfp REQUIRED)
# Some SZ3 installs generate SZ3/version.hpp under a separate build include path.
set(SZ3_EXTRA_INCLUDE_DIR "" CACHE PATH "Optional SZ3 generated-header directory")
function(exactz_target target source)
  add_executable(${target} ${source})
  target_include_directories(${target} PRIVATE ${SZ3_INCLUDE_DIR} ${SZ3_EXTRA_INCLUDE_DIR} ${ZSTD_INCLUDE_DIR} ${ZFP_INCLUDE_DIR})
  target_link_libraries(${target} PRIVATE CUDA::cudart ${ZSTD_LIBRARY} ${ZFP_LIBRARY} OpenMP::OpenMP_CXX)
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda> $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-fopenmp>)
  install(TARGETS ${target} RUNTIME DESTINATION bin)
endfunction()
if(EXACTZ_BUILD_SINGLE)
  # These backends are included unconditionally by the original single-GPU source.
  find_path(MGARD_INCLUDE_DIR mgard/compress_x.hpp REQUIRED)
  find_library(MGARD_LIBRARY NAMES mgard REQUIRED)
  find_path(CUSZP_INCLUDE_DIR cuSZp.h REQUIRED)
  find_library(CUSZP_LIBRARY NAMES cuSZp REQUIRED)
  exactz_target(EXaCTz_single src/ExtreM_cuda.cu)
  target_include_directories(EXaCTz_single PRIVATE ${MGARD_INCLUDE_DIR} ${CUSZP_INCLUDE_DIR})
  target_link_libraries(EXaCTz_single PRIVATE ${MGARD_LIBRARY} ${CUSZP_LIBRARY})
endif()
if(EXACTZ_BUILD_MPI)
  find_package(MPI REQUIRED COMPONENTS CXX)
  exactz_target(EXaCTz_mpi src/pExtreMz_all_order.cu)
  target_link_libraries(EXaCTz_mpi PRIVATE MPI::MPI_CXX)
endif()
